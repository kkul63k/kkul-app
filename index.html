const day = String(date.getDate()).padStart(2, '0'); const hours = String(date.getHours()).padStart(2, '0'); const minutes = String(date.getMinutes()).padStart(2, '0'); const seconds = String(date.getSeconds()).padStart(2, '0'); return `${year}${month}${day}T${hours}${minutes}${seconds}Z`; }; /** * 모든 적금 상품들을 iCalendar (.ics) 형식으로 생성 */ const generateICalendar = () => { const products = JSON.parse(localStorage.getItem('products') || '[]'); const today = new Date(); today.setHours(0, 0, 0, 0); // iCalendar 헤더 let icsContent = `BEGIN:VCALENDAR VERSION:2.0 PRODID:-//merryon//savings schedule//KO CALSCALE:GREGORIAN METHOD:PUBLISH X-WR-CALNAME:적금 일정표 X-WR-TIMEZONE:Asia/Seoul X-WR-CALDESC:꿀단지 자산관리 - 적금 납입 및 만기 일정 DTSTAMP:${formatDateForICS(new Date())} `; // 모든 VEVENT 생성 const allEvents = []; products.forEach(product => { // ===== 예금 처리 ===== if ((product.productType || product.type) === '예금') { const maturityDate = new Date(product.extendedMaturityDate || product.maturityDate); maturityDate.setHours(0, 0, 0, 0); // 예금 만기일 이벤트 const event = { uid: `${product.id}-maturity@merryon.local`, dtstamp: formatDateForICS(new Date()), dtstart: `${maturityDate.getFullYear()}${String(maturityDate.getMonth() + 1).padStart(2, '0')}${String(maturityDate.getDate()).padStart(2, '0')}`, summary: `${product.product} - 만기 (${parseInt(product.amount || 0).toLocaleString()}원)`, description: `은행: ${product.bank} 상품: ${product.product} 금액: ${parseInt(product.amount || 0).toLocaleString()}원 만기일입니다.`, category: 'MATURITY' }; allEvents.push(event); return; } // ===== 적금 처리 ===== if ((product.productType || product.type) !== '적금') { return; } // 1일 적금은 제외 if ((product.paymentPeriod || '개월') === '일') { return; } let startDate = new Date(product.startDate || product.savedDate); if (isNaN(startDate.getTime())) { startDate = new Date(today); } startDate.setHours(0, 0, 0, 0); const maturityDate = new Date(product.extendedMaturityDate || product.maturityDate); maturityDate.setHours(0, 0, 0, 0); const paymentPeriod = product.paymentPeriod || '개월'; const paymentCycleNum = parseInt(product.paymentCycle || 1) || 1; const monthlyAmount = parseInt(product.monthlyPaymentAmount || product.depositAmount || 0) || 0; let currentPaymentDate = new Date(startDate); let cycleCount = 1; // 각 납입일별 이벤트 생성 while (currentPaymentDate <= maturityDate && cycleCount <= paymentCycleNum) { if (currentPaymentDate >= today) { const event = { uid: `${product.id}-payment-${cycleCount}@merryon.local`, dtstamp: formatDateForICS(new Date()), dtstart: `${currentPaymentDate.getFullYear()}${String(currentPaymentDate.getMonth() + 1).padStart(2, '0')}${String(currentPaymentDate.getDate()).padStart(2, '0')}`, summary: `${product.product} - ${monthlyAmount.toLocaleString()}원 납입 (${cycleCount}회차)`, description: `은행: ${product.bank} 상품: ${product.product} 금액: ${monthlyAmount.toLocaleString()}원 ${cycleCount}회차 정기 납입일입니다.`, category: 'PAYMENT' }; allEvents.push(event); } // 다음 납입일 계산 if (paymentPeriod === '일') { currentPaymentDate.setDate(currentPaymentDate.getDate() + 1); } else if (paymentPeriod === '주') { currentPaymentDate.setDate(currentPaymentDate.getDate() + 7); } else if (paymentPeriod === '개월') { currentPaymentDate.setMonth(currentPaymentDate.getMonth() + 1); } cycleCount++; } // 만기일 이벤트 const event = { uid: `${product.id}-maturity@merryon.local`, dtstamp: formatDateForICS(new Date()), dtstart: `${maturityDate.getFullYear()}${String(maturityDate.getMonth() + 1).padStart(2, '0')}${String(maturityDate.getDate()).padStart(2, '0')}`, summary: `${product.product} - 만기 (${monthlyAmount.toLocaleString()}원)`, description: `은행: ${product.bank} 상품: ${product.product} 만기일입니다.`, category: 'MATURITY' }; allEvents.push(event); }); // 이벤트를 날짜순으로 정렬 allEvents.sort((a, b) => { const dateA = a.dtstart; const dateB = b.dtstart; return dateA.localeCompare(dateB); }); // 각 이벤트를 iCalendar 형식으로 추가 allEvents.forEach(event => { icsContent += `BEGIN:VEVENT UID:${event.uid} DTSTAMP:${event.dtstamp} DTSTART;VALUE=DATE:${event.dtstart} DTEND;VALUE=DATE:${String(parseInt(event.dtstart) + 1).padStart(8, '0')} SUMMARY:${event.summary} DESCRIPTION:${event.description} CATEGORIES:${event.category} STATUS:CONFIRMED SEQUENCE:0 END:VEVENT `;
